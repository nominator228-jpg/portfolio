# comfy_client.py — очередь в ComfyUI и ожидание результата
def queue(url, graph):
    r = requests.post(f"{url}/prompt",
        json={"prompt": graph, "client_id": str(uuid.uuid4())}, timeout=30)
    data = r.json()
    if data.get("error") or data.get("node_errors"):
        raise RuntimeError(f"ComfyUI error: {data}")
    return data["prompt_id"]

def wait(url, prompt_id, timeout, interval=5):
    start = last_contact = time.time()
    while time.time() - start < timeout:
        job = poll_history(url, prompt_id)
        if job: last_contact = time.time()
        if time.time() - last_contact > UNREACHABLE_AFTER:
            raise RuntimeError("pod stopped or crashed")
        if job and job["status"].get("status_str") == "error":
            raise RuntimeError(error_details(job))  # какая нода упала и почему
        if job and job["status"].get("completed"):
            return job
        time.sleep(interval)
